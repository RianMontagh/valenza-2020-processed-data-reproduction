cmake_minimum_required(VERSION 3.16)
project(avulsion_processed LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(GDAL REQUIRED)
add_executable(fingerprint src/fingerprint.cpp)
target_include_directories(fingerprint PRIVATE ${GDAL_INCLUDE_DIRS})
target_link_libraries(fingerprint PRIVATE ${GDAL_LIBRARIES} stdc++fs)
target_compile_options(fingerprint PRIVATE -O3 -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME analytical_limits COMMAND fingerprint selftest)
